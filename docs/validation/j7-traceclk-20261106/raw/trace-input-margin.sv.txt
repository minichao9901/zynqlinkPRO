`timescale 1ps/1ps
module trace_input_margin;
    reg sample_clk=0;
    reg clk=0;
    reg rst=1;
    reg trace_clk=0;
    reg [3:0] data=0;
    always #2500 sample_clk=~sample_clk;
    always #4167 clk=~clk;

    orbtrace_parallel #(.RAW_DIAGNOSTIC(4),.USE_INPUT_DELAY(0)) dut(
        .trace_clk(trace_clk),.trace_io_clk(trace_clk),.trace_pad_clk(trace_clk),
        .sample_clk(sample_clk),.trace_data(data),.width(2'd3),.clk(clk),.rst(rst),
        .ready(1'b1),.diagnostic_reset(rst),.ddr_hold(1'b0),.ddr_sampler_gate(1'b1),
        .ddr_parser_enable(1'b1),.ddr_parser_clear(1'b0));

    function automatic [3:0] nibble(input integer index);
        nibble = (index*11) ^ (index>>2) ^ (index>>5);
    endfunction

    integer mhz, half, skew, phase, count=0, errors=0;
    reg seen=0;
    reg [7:0] expected;
    always @(negedge sample_clk) begin
        if (dut.raw_oversample_filtered.byte_toggle != seen) begin
            seen=dut.raw_oversample_filtered.byte_toggle;
            expected={nibble(2*count+1),nibble(2*count)};
            if(dut.raw_oversample_filtered.sampled_byte !== expected)errors=errors+1;
            count=count+1;
        end
    end

    initial begin
        if (!$value$plusargs("mhz=%d",mhz) || !$value$plusargs("skew=%d",skew) || !$value$plusargs("phase=%d",phase))
            $fatal(1,"mhz, skew and phase required");
        half=(500000+mhz/2)/mhz;
        #50000; rst=0;
        #(100000+phase);
        fork
            begin
                #(half/2);
                for (integer i=0;i<512;i=i+1) begin
                    trace_clk=~trace_clk;
                    #(half);
                end
            end
            begin
                for (integer i=0;i<512;i=i+1) begin
                    data[3:1]=3'(nibble(i)>>1);
                    #(half);
                end
            end
            begin
                #(skew);
                for (integer i=0;i<512;i=i+1) begin
                    data[0]=1'(nibble(i));
                    #(half);
                end
            end
        join
        #100000;
        $display("RESULT mhz=%0d skewPs=%0d phasePs=%0d bytes=%0d badBytes=%0d",mhz,skew,phase,count,errors);
        $finish;
    end
endmodule
