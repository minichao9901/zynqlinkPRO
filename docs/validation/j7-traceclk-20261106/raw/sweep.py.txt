"""Reversible F103ZE TRACECLK sweep with one fixed SRAM-only workload."""
import argparse
from contextlib import redirect_stdout
import hashlib
import json
from pathlib import Path
import sys
import time

import usb.util

WORK = Path(__file__).resolve().parent
ROOT = WORK.parents[1]
sys.path[:0] = [str(ROOT / 'tools'), str(ROOT / 'etm-service/tools')]
from benchmark_rtt_forward import core_read, core_write
from dap_pipeline_bench import cmd, open_dap, setup
from decode_stream import decode_stream
from engine_profile import activate_native_profile
from f103_ram_image import plan
from probe_ownership import Ownership
from probe_stream import Probe
from run_f103_clock_fixture import APB1ENR, CFGR, CR, FLASH, hsi, restore, wait_bits
from run_f103_ram_fixture import flash_plan, verify
import test_orbtrace_board
from test_probe_board import Memory
from validate_fclk_performance import physical_snapshot

SAFE = 10000
SRAM = 0x20000000
DHCSR = 0xe000edf0
DCRDR = 0xe000edf8
VTOR = 0xe000ed08
DEMCR = 0xe000edfc
DWT = 0xe0001000
FPB = 0xe0002000
CORE_REGS = [*range(19), 20]
CLOCK_REGS = dict(CR=CR, CFGR=CFGR, APB1ENR=APB1ENR, FLASH_ACR=FLASH)
PAIR = ROOT / 'prebuilt/j7-adapter-20261106'
ELF = WORK / 'trace-workload.elf'


def require(check, message):
    if not check:
        raise RuntimeError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def write_json(path, data):
    path.write_text(json.dumps(data, indent=2) + '\n', encoding='utf8')


def set_hclk(mem, baseline, trace_mhz):
    hclk = trace_mhz * 2
    prediv = 1 if hclk % 8 == 0 else 2
    mul = hclk // (8 // prediv)
    require(24 <= trace_mhz <= 36 and 2 <= mul <= 16 and hclk == mul * (8 // prediv), 'Invalid in-spec PLL point')
    mem.write_word(FLASH, (baseline['FLASH_ACR'] & ~7) | 2)
    mem.write_word(APB1ENR, baseline['APB1ENR'] & ~(1 << 23))
    hsi(mem)
    mem.write_word(CR, mem.word(CR) & ~(1 << 24))
    wait_bits(mem, CR, 1 << 25, 0)
    require(mem.word(CR) & (1 << 17), 'HSE is not ready')
    # AHB /1, APB1 /2, APB2 /1 and ADC /8 throughout the experiment.
    config = (baseline['CFGR'] & ~0x3fffff) | (4 << 8) | (3 << 14) | (1 << 16)
    config |= ((prediv == 2) << 17) | ((mul - 2) << 18)
    mem.write_word(CFGR, config)
    mem.write_word(CR, mem.word(CR) | (1 << 24))
    wait_bits(mem, CR, 1 << 25, 1 << 25)
    mem.write_word(CFGR, config | 2)
    current = wait_bits(mem, CFGR, 12, 8)
    require(current == (config | 10), f'Unexpected clock readback {current:#x}')
    return dict(hclkHz=hclk*1_000_000, traceClkHz=trace_mhz*1_000_000, rccCr=hex(mem.word(CR)),
                rccCfgr=hex(current), pllInputHz=8_000_000//prediv, pllMultiplier=mul)


def cycle_check(mem, hclk):
    def read_count():
        begin = time.perf_counter()
        count = mem.word(DWT + 4)
        return (begin + time.perf_counter()) / 2, count

    mem.write_word(DHCSR, 0xa05f0001)
    try:
        start, first = read_count()
        time.sleep(.2)
        end, last = read_count()
    finally:
        mem.write_word(DHCSR, 0xa05f0003)
    cycles = (last - first) & 0xffffffff
    frequency = cycles / (end - start)
    require(abs(frequency / hclk - 1) < .02, f'DWT CPU clock differs: {frequency}')
    return dict(cycles=cycles, hostSeconds=end-start, hclkHz=frequency)


def run():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument('--diagnostic', action='store_true', help='24 MHz reference and 36 MHz GPIO comparison')
    mode.add_argument('--max-window', action='store_true', help='32 MHz full windows and a 100 MiB ring')
    mode.add_argument('--ring-only', action='store_true', help='32 MHz 100 MiB hardware/USB comparison')
    mode.add_argument('--usb-compare', action='store_true', help='Alternate equal-size ring downloads at 24 and 32 MHz')
    args = parser.parse_args()
    result = WORK / ('diagnostic.json' if args.diagnostic else 'max-window.json' if args.max_window else
                     'ring-only.json' if args.ring_only else 'usb-compare.json' if args.usb_compare else 'sweep.json')
    require(not result.exists(), 'Use fresh sweep evidence')
    image = plan(ELF)
    trap = image['symbols']['trace_trap'] & ~1
    expected_flash = flash_plan(ROOT / 'prebuilt/golden/f103ze-cdc48.elf')
    report = dict(rows=[], fixtureSHA256=sha(ELF.read_bytes()), gpioSpeed=3, restored=False)
    activate_native_profile()
    lease = Ownership()
    dev = mem = initial = saved = core = fpb = sram = debug = None

    def connect():
        nonlocal dev, mem
        dev, tx, rx = open_dap()
        setup(tx, rx, SAFE)
        mem = Memory(tx, rx)

    def close():
        nonlocal dev
        if dev is not None:
            usb.util.dispose_resources(dev)
            dev = None

    def point(trace_mhz, ms, round_number, gpio=3, physical=False, flight=False):
        row = dict(traceMHz=trace_mhz, milliseconds=ms, round=round_number, gpio=gpio, passed=False)
        report['rows'].append(row)
        prefix = f'diag-gpio{gpio}-' if args.diagnostic else 'usbcompare-ring-' if args.usb_compare else 'ring-' if flight else ''
        folder = WORK / f'{prefix}trace{trace_mhz}-{ms}ms-r{round_number}'
        require(not folder.exists(), 'Capture output already exists')
        row['clock'] = set_hclk(mem, saved, trace_mhz)
        core_write(mem, 15, image['entry'])
        row['DWT'] = cycle_check(mem, row['clock']['hclkHz'])
        require((image['entry'] & ~1) <= core_read(mem, 15) < trap, 'SRAM fixture did not run in expected loop')
        close()
        previous_argv = sys.argv
        started = time.perf_counter()
        try:
            sys.argv = ['test_orbtrace_board', '--ram-image', '--target-elf', str(ELF),
                        '--bit', str(PAIR/'probe.bit'), '--elf', str(PAIR/'application.elf'),
                        '--output', str(folder), '--milliseconds', str(ms),
                        '--speed-khz', str(SAFE), '--gpio-speed', str(gpio), '--transport', 'ddr']
            if physical:
                sys.argv.append('--verify-ddr-physical')
            if flight:
                sys.argv += ['--flight', '--ring-bytes', str(100<<20), '--trigger', 'after',
                             '--trigger-after', '6', '--arm-timeout', '15']
            with (WORK / (folder.name + '.log')).open('w', encoding='utf8') as log, redirect_stdout(log):
                test_orbtrace_board.main()
        except Exception as error:
            row['captureError'] = repr(error)
        finally:
            sys.argv = previous_argv
        row['captureHostSeconds'] = time.perf_counter() - started
        connect()
        halt = mem.word(DHCSR)
        require(halt & (1 << 17) and not halt & (1 << 25), f'Fixture did not remain halted without reset: {halt:#x}')
        a = json.loads((folder/'acceptance.json').read_text(encoding='utf8'))
        require(a.get('restored') and a.get('fpb_restored') and not a.get('target_cleanup_error'), 'Per-capture CoreSight cleanup failed')
        ddr = json.loads((folder/'ddr/result.json').read_text(encoding='utf8'))
        require(ddr.get('usbOwnerReleased') and not ddr.get('cleanupError'), 'DDR owner did not release')
        row.update(quality=ddr.get('quality'), capturePassed=a.get('capture_passed',False),
                   formatterSHA256=sha((folder/'ddr/capture.frm1').read_bytes()))
        if physical:
            require(ddr.get('usbMatchesPhysical'), 'Physical DDR readback differs from USB')
            row['usbMatchesPhysical']=True
            row['physicalSHA256']=ddr['physicalSha256']
        if row['capturePassed']:
            data = (folder/'etm.bin').read_bytes()
            row['etmSHA256'] = sha(data)
            if flight:
                row['passed'] = row['quality']['requestedWindowComplete'] and row['quality']['historyOverwritten']
            else:
                t = time.perf_counter()
                decoded = decode_stream(data, ELF, folder, a['config'])
                row['decodeSeconds'] = time.perf_counter()-t
                row['summary'] = decoded['summary']
                row['passed'] = bool(row['quality']['requestedWindowComplete'] and
                    decoded['summary']['packetIssues'] == 0 and decoded['summary']['instructions'] > 0 and
                    {'trace_loop','trace_mix'} <= decoded['summary']['functions'].keys())
        write_json(result, report)
        print(json.dumps(row), flush=True)
        return row['passed']

    try:
        state = physical_snapshot()
        p = Probe()
        try:
            streams = [p.action(c, 2) for c in (0x31, 0x32)]
        finally:
            p.close()
        require(state['allObservedEnginesIdle'] and all(rc == 0 and not words[0]&1 for rc, words in streams), 'Probe occupied')
        connect()
        require(mem.word(0xe0042000)&0xfff == 0x414, 'F103ZE required')
        report['flashBefore'] = verify(mem, expected_flash)
        original = mem.word(DHCSR)
        mem.write_word(DHCSR, 0xa05f0003)
        initial = original
        require(mem.word(DHCSR)&(1 << 17), 'Target did not halt')
        dcrdr = mem.word(DCRDR)
        core = {n:core_read(mem,n) for n in CORE_REGS}
        require(not core[16]&0x1ff, 'Target halted in an exception handler')
        saved = {k:mem.word(a) for k,a in CLOCK_REGS.items()}
        require(saved['CFGR'] == 0x51040a and saved['CR'] & (1<<17), 'Expected original CDC48 clock')
        fpb = mem.read(FPB, 40)
        sram = mem.read(SRAM, 65536)
        require(mem.read(SRAM,65536) == sram,'SRAM changed while halted')
        (WORK/(result.stem+'-sram-private.bin')).write_bytes(sram)
        (WORK/(result.stem+'-core-private.json')).write_text(json.dumps(core), encoding='utf8')
        debug = dict(dcrdr=dcrdr, vtor=mem.word(VTOR), demcr=mem.word(DEMCR))
        mem.write_word(DEMCR, debug['demcr'] | (1 << 24))
        debug.update(dwt=mem.word(DWT), cycles=mem.word(DWT+4))
        report.update(initialDHCSR=hex(initial), clockBefore=saved, sramSHA256=sha(sram), fpbSHA256=sha(fpb))
        write_json(result, report)
        mem.write_word(DWT, debug['dwt'] | 1)
        mem.write_word(FPB, 2)
        for address, data in image['payloads']:
            mem.write(address, data)
        verify(mem, image['payloads'])
        mem.write_word(VTOR, SRAM)
        core_write(mem, 20, 1)
        core_write(mem, 17, image['stack'])
        core_write(mem, 13, image['stack'])
        core_write(mem, 16, 0x01000000)

        if args.usb_compare:
            for n in (1,2,3):
                for mhz in (24,32):
                    require(point(mhz,100,n,flight=True),'Equal-size ring comparison failed')
        elif args.ring_only:
            report['ring32MHz'] = point(32,100,1,flight=True)
        elif args.max_window:
            report['fullWindows32MHz'] = [point(32,2000,n) for n in (1,2,3)]
            report['ring32MHz'] = point(32,100,1,flight=True)
        elif args.diagnostic:
            report['reference24MHz'] = [point(24,1000,n) for n in (1,2,3)]
            for gpio in (1,2,3):
                checks = [point(36,100,n,gpio,physical=(gpio==3 and n==1)) for n in (1,2,3)]
                if all(checks):
                    repeat = [point(36,1000,n,gpio) for n in (1,2,3)]
                    if all(repeat):
                        report['confirmed36MHzGpio'] = gpio
        else:
            require(point(24,100,1), '24 MHz control did not pass')
            best = 24
            for mhz in (28,32,36):
                if point(mhz,100,1):
                    best = mhz
                elif mhz in (28,32):
                    if point(mhz-2,100,1):best=mhz-2
                    break
                else:
                    break
            report['shortWindowBestMHz'] = best
            report['longWindowChecks'] = [point(best,1000,n) for n in (1,2,3)]
            report['confirmedMHz'] = best if all(report['longWindowChecks']) else None
    finally:
        try:
            if initial is not None:
                if dev is None:connect()
                mem.write_word(DHCSR,0xa05f0003)
                require(mem.word(DHCSR)&(1<<17), 'Cannot halt for restoration')
                if saved is not None:
                    restore(mem, saved)
                    after = {k:mem.word(a) for k,a in CLOCK_REGS.items()}
                    report['clockAfter'] = after
                    require(after==saved,'Target clock restore mismatch')
                if sram is not None:
                    current = mem.read(SRAM,len(sram))
                    for offset in range(0,len(sram),1024):
                        if current[offset:offset+1024]!=sram[offset:offset+1024]:
                            mem.write(SRAM+offset,sram[offset:offset+1024])
                    require(mem.read(SRAM,len(sram))==sram,'Full SRAM restore mismatch')
                    report['wholeSramRestored']=True
                if fpb is not None:
                    mem.write(FPB+4,fpb[4:])
                    mem.write_word(FPB,2|(int.from_bytes(fpb[:4],'little')&1))
                    require(mem.read(FPB,40)==fpb,'FPB restore mismatch')
                if debug is not None:
                    mem.write_word(VTOR,debug['vtor'])
                    mem.write_word(DWT+4,debug['cycles'])
                    mem.write_word(DWT,debug['dwt'])
                    require(mem.word(VTOR)==debug['vtor'] and mem.word(DWT)==debug['dwt'] and mem.word(DWT+4)==debug['cycles'], 'VTOR or DWT restore mismatch')
                    mem.write_word(DEMCR,debug['demcr'])
                    require(mem.word(DEMCR)==debug['demcr'],'DEMCR restore mismatch')
                if core is not None:
                    # Restore banked SP and special controls before R13 and PC.
                    for n in (17,18,20,*range(17)):
                        core_write(mem,n,core[n])
                    require({n:core_read(mem,n) for n in CORE_REGS}==core,'Core register restore mismatch')
                    report['coreRestored']=True
                if debug is not None:mem.write_word(DCRDR,debug['dcrdr'])
                report['flashAfter']=verify(mem,expected_flash)
                mem.write_word(DHCSR,0xa05f0000|(initial&0x0b))
                final=mem.word(DHCSR)
                require((initial^final)&(1<<17)==0 and not final&(1<<25),'Original run/halt state not restored')
                report.update(finalDHCSR=hex(final),restored=True)
                cmd(mem.out,mem.inp,[3])
        finally:
            close()
            lease.close()
            write_json(result, report)


if __name__ == '__main__':
    run()
