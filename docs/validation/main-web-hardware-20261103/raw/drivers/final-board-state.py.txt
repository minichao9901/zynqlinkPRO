from pathlib import Path
import sys,json,hashlib,datetime
sys.path[:0]=[str(Path.cwd()),'tools']
from validate_fclk_performance import physical_snapshot
from probe_build_info import live_clock_snapshot
from dap_pipeline_bench import open_dap,setup
from test_probe_board import Memory
from target_snapshot import snapshot
from probe_stream import Probe
import usb.util
work=Path('build/main-web-hardware-20261103')
r=dict(recordedAtUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(),probeQspiWrites=False)
p=Probe()
try:
 for cmd,name in [(0x31,'rtt'),(0x32,'hss')]:
  rc,words=p.action(cmd,2);assert rc==0 and not words[0]&1
  r[name]=dict(stopped=True,rawStatus=words)
finally:p.close()
d,tx,rx=open_dap();m=Memory(tx,rx)
try:
 setup(tx,rx,10000);actual=m.read(0x08000000,524288);expected=(work/'original-flash.bin').read_bytes();assert actual==expected
 r['originalFlash']=dict(bytes=len(actual),sha256=hashlib.sha256(actual).hexdigest(),exact=True)
 r['target']=snapshot(m);r['DHCSR']=hex(m.word(0xe000edf0));r['VTOR']=hex(m.word(0xe000ed08))
 assert not int(r['DHCSR'],16)&(1<<17) and r['target']['registers']['0x40021004']=='0x51040a'
finally:usb.util.dispose_resources(d)
r['probePhysical']=physical_snapshot();assert r['probePhysical']['allObservedEnginesIdle']
r['clock']=live_clock_snapshot();assert abs(r['clock']['actual_link_Hz']-10000000)<10
r['probePair']='prebuilt/main-ddr-usb-133-statfix-20261104'
r['targetRunning']='original full Flash / F103 CDC48; external NOR/EEPROM last verified original before switching from HAL, no external writes since'
r['passed']=True;(work/'final-board-state.json').write_text(json.dumps(r,indent=2)+'\n');print('Final full Flash exact, CDC48 running, RTT/HSS/DDR/DMA idle, actual SWD10/FCLK120')
