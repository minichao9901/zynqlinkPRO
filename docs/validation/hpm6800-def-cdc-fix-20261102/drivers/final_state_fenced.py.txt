import sys,json,re
from pathlib import Path
from datetime import datetime,timezone
sys.path.insert(0,'tools')
import usb.util
from dap_pipeline_bench import open_dap,drain,xfer,le
from probe_stream import Probe
from benchmark_hpm5301 import clock
from validate_fclk_performance import physical_snapshot

base=Path('build/hpm6800-def-cdc-fix-20261102')
observed=json.loads((base/'final-web-idcode.json').read_text())
log=observed['log']['text']
assert '0x1000563D' in log
dmstatus=int(re.search(r'dmstatus\s*[=:]\s*(0x[0-9a-f]+)',log,re.I)[1],16)
assert dmstatus&(1<<10) and not dmstatus&(1<<9)
report=dict(time_UTC=datetime.now(timezone.utc).isoformat(),before=physical_snapshot(),
    dmstatus=hex(dmstatus),dmstatus_source='Final frozen EXE FlashView readIdcode log before configuration-only calls',
    operation='Probe configuration only: connect/clock/idle/sampling/disconnect, no DMI/SBA memory operation')
assert report['before']['allObservedEnginesIdle']
dev,out,inp=open_dap()
probe=Probe()
try:
    report['services']={hex(c):probe.action(c,2) for c in (0x31,0x32)}
    assert all(rc==0 and not info[0]&1 for rc,info in report['services'].values())
    drain(inp)
    commands=[[2,2],[0x11]+le(10000000),[4,8,64,0,0,0],[0x89,0],[3]]
    report['configuration_replies']=[xfer(out,inp,c).hex()for c in commands]
    assert [r[:6]if i==3 else r for i,r in enumerate(report['configuration_replies'])]==['0202','1100','0400','890000','0300'],report
    report['clock']=clock(out,inp)
    report['idle_query']=xfer(out,inp,[0x8d,255]).hex()
    query=bytes.fromhex(report['idle_query'])
    assert query[:3]==b'\x8d\x00\x08' and query[3] in (15,64),report
    report['idle_limit_note']='Normal raw JTAG context reports 15; virtual RISC-V memory backend reports 64. No backend activation performed in final cleanup.'
finally:
    probe.close()
    usb.util.release_interface(dev,0)
    usb.util.dispose_resources(dev)
report['after']=physical_snapshot()
assert report['after']['allObservedEnginesIdle']
report['passed']=True
(base/'final-state-fenced.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
