"""Read-only byte/CRC counters at COM reads and completed native read RPCs."""
import sys,runpy,threading,time,json,base64,zlib,ctypes
from pathlib import Path
import serial,serial.win32 as win32
base=Path(__file__).resolve().parent;sys.path.insert(0,str(Path.cwd()/'etm-service/tools'))
from native_probe import NativeProbe
lock=threading.Lock();ports={};counter=0
original_read=serial.Serial.read;original_call=NativeProbe.serial_call;original_error=win32.ClearCommError
errors=[];max_queued=0
def check_errors(handle,flags,stat):
    global max_queued
    ok=original_error(handle,flags,stat)
    if ok:
        bits=ctypes.cast(flags,ctypes.POINTER(win32.DWORD)).contents.value;n=ctypes.cast(stat,ctypes.POINTER(win32.COMSTAT)).contents.cbInQue
        with lock:
            max_queued=max(max_queued,n)
            if bits:errors.append(dict(at=time.time(),flags=bits,queued=n))
    return ok
win32.ClearCommError=check_errors
def read(port,n=1):
    key=getattr(port,'_diagnostic_key','unknown')
    with lock:
        row=ports.setdefault(key,dict(comBytes=0,comCrc=0,rpcBytes=0,rpcCrc=0,active=0,maxActive=0,reads=0))
        row['active']+=1;row['maxActive']=max(row['maxActive'],row['active'])
    try:
        data=original_read(port,n)
        with lock:row['comBytes']+=len(data);row['comCrc']=zlib.crc32(data,row['comCrc']);row['reads']+=1
        return data
    finally:
        with lock:row['active']-=1
serial.Serial.read=read
def call(proxy,action,value):
    global counter
    key=(value.get('owner'),value.get('path'));port=proxy.ports.get(key)
    result=original_call(proxy,action,value)
    if action=='open':
        with lock:
            counter+=1
            ports[str(counter)]=dict(comBytes=0,comCrc=0,rpcBytes=0,rpcCrc=0,active=0,maxActive=0,reads=0)
            proxy.ports[key]._diagnostic_key=str(counter)
    if action=='read':
        data=base64.b64decode(result['data']);name=getattr(port,'_diagnostic_key','unknown')
        with lock:row=ports[name];row['rpcBytes']+=len(data);row['rpcCrc']=zlib.crc32(data,row['rpcCrc'])
        config=base/'serial-read-delay.json'
        if data and config.exists():time.sleep(min(.2,max(0,json.loads(config.read_text())['seconds'])))
    return result
NativeProbe.serial_call=call
def persist():
    while True:
        with lock:out=dict(ports={k:dict(v)for k,v in ports.items()},maxQueued=max_queued,errors=list(errors))
        (base/'serial-layer-observer.json').write_text(json.dumps(out,indent=2)+'\n');time.sleep(.5)
threading.Thread(target=persist,daemon=True).start()
sys.argv=['toolbox.py',*sys.argv[1:]];runpy.run_path('toolbox.py',run_name='__main__')
