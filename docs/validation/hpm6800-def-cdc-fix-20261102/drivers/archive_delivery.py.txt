def memory_hashes(value):
    if isinstance(value,list):return [memory_hashes(v)for v in value]
    if isinstance(value,dict):
        out={k:memory_hashes(v)for k,v in value.items()if k!='$bytes'}
        if '$bytes'in value:
            data=bytes(value['$bytes']);out['$bytesEvidence']=dict(bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
        return out
    return value
def copy(p,q):
    assert not any(t in p.read_bytes()for t in tokens),('Runtime token',p)
    q.parent.mkdir(parents=True,exist_ok=True)
    if p.name=='steps.json':
        content=load(p);changed=False
        for row in content:
            if row.get('request',{}).get('method')in ('memRead','memWrite'):
                row.update(memory_hashes(row));changed=True
        if changed:
            save(q,content);redactions[q.relative_to(D).as_posix()]=dict(original_sha256=sha(p),reason='Debugger temporary SRAM byte arrays retain length/SHA only; full source evidence remains in build')

    shutil.copy2(p,q)
def zipped(p,q):
