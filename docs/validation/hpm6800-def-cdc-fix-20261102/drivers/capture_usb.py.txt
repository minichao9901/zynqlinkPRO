import subprocess,time,json,sys
from pathlib import Path
b=Path(__file__).resolve().parent;name=sys.argv[1];out=b/(name+'.pcap');stop=b/(name+'.stop')
assert not stop.exists()
with (b/(name+'-pcap.log')).open('w')as log:
    proc=subprocess.Popen(['C:/Program Files/USBPcap/USBPcapCMD.exe','-d',r'\\.\USBPcap3','-o',str(out),'--devices','8','-b','33554432'],stdin=subprocess.PIPE,stdout=log,stderr=subprocess.STDOUT,creationflags=subprocess.CREATE_NO_WINDOW)
    (b/(name+'-pcap-process.json')).write_text(json.dumps(dict(pid=proc.pid,command=proc.args)))
    while proc.poll() is None and not stop.exists():time.sleep(.2)
    if proc.poll() is None:
        # GUI-subsystem USBPcapCMD uses ReadConsoleInput for 'q'; a hidden
        # pipe is not a console. Stop only this PID after the stream is quiet.
        # Parser must explicitly report any incomplete final pcap record.
        proc.terminate();proc.wait(3)
    print(proc.returncode,flush=True)
