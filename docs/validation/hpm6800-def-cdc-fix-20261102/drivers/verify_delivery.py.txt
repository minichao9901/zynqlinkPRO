from pathlib import Path
import json,hashlib,subprocess,gzip
from elftools.elf.elffile import ELFFile
ROOT=Path.cwd();B=ROOT/'build/hpm6800-def-cdc-fix-20261102';D=ROOT/'docs/validation/hpm6800-def-cdc-fix-20261102'
P=ROOT/'prebuilt/hpm6800-usb-fixed-133-20261102';T=ROOT/'prebuilt/targets/hpm6800-usb-ram-20261102'
def sha(data):return hashlib.sha256(data).hexdigest()
def load(p):return json.loads(p.read_text(encoding='utf-8'))
checks={}
for root,manifest in ((P,'manifest.json'),(T,'manifest.json'),(D,'evidence.json')):
    for name,entry in load(root/manifest)['files'].items():checks[(root/name).relative_to(ROOT).as_posix()]=entry['sha256']
for filename,expected in checks.items():assert sha((ROOT/filename).read_bytes())==expected,filename
frozen=load(B/'frozen-package-evidence.json')
assert sha((ROOT/'dist/zynq-etm-bridge/zynq-etm-bridge.exe').read_bytes())==frozen['exe_sha256']
assert sha((ROOT/'dist/zynq-etm-bridge/package-manifest.json').read_bytes())==frozen['package_manifest_sha256']
build=load(P/'ps-build.json');assert sha((P/'application.elf').read_bytes())==build['elf_sha256']
for name,expected in build['source_sha256'].items():assert sha((ROOT/'firmware/xusbps_daplink_dma'/name).read_bytes())==expected,name
for name in ('ahb16','axi32'):
    with (T/(name+'.elf')).open('rb')as f:assert ELFFile(f).get_section_by_name('.text').data()==(T/(name+'.bin')).read_bytes()
for path in (D/'pcap').glob('*.gz'):assert gzip.decompress(path.read_bytes())==(B/path.name[:-3]).read_bytes()
for filename in ['prebuilt/golden/probe.bit','prebuilt/golden/application.elf','prebuilt/hpm6800-web-133-20261102/probe.bit','prebuilt/hpm6800-web-133-20261102/application.elf']:
    assert (ROOT/filename).read_bytes()==subprocess.check_output(['git','show','HEAD:'+filename]),filename
for source,packed in [('etm-service/tools/native_serial.py','backend/tools/native_serial.py'),('etm-service/tools/native_probe.py','backend/tools/native_probe.py'),('web/app/core/native-probe.js','toolbox/app/core/native-probe.js')]:
    assert (ROOT/source).read_bytes()==(ROOT/'dist/zynq-etm-bridge/_internal'/packed).read_bytes(),source
batch=subprocess.Popen(['git','cat-file','--batch'],stdin=subprocess.PIPE,stdout=subprocess.PIPE)
for filename,expected in checks.items():
    batch.stdin.write((':'+filename+'\n').encode());batch.stdin.flush();header=batch.stdout.readline().decode().split()
    assert len(header)==3 and header[1]=='blob',('Not staged',filename,header)
    data=batch.stdout.read(int(header[2]));assert batch.stdout.read(1)==b'\n';assert sha(data)==expected,('Staged SHA differs',filename)
batch.stdin.close();assert batch.wait()==0
files=subprocess.check_output(['git','diff','--cached','--name-only','-z']).decode().split('\0')
tokens=[p.read_bytes().strip()for p in B.rglob('cli-token')]
for filename in files:
    if not filename:continue
    assert not filename.startswith(('build/','dist/'))and 'target-backup'not in filename and 'readback-private'not in filename,filename
    assert not any(t in (ROOT/filename).read_bytes()for t in tokens),filename
for path in (D/'raw').rglob('steps.json'):
    for row in load(path):
        if row.get('request',{}).get('method')in ('memRead','memWrite'):assert '"$bytes"'not in json.dumps(row),path
print('Verified',len(checks),'manifest-bound staged files, frozen source bytes and both target machine-code variants; baselines unchanged; no private backups/tokens/build/dist staged.')
