from pathlib import Path
import hashlib,json,subprocess,sys,shutil
root=Path(__file__).resolve().parents[2]
out=root/'build/fclk-resume-20261101/swd-fixed133/sim';out.mkdir(parents=True,exist_ok=False)
golden=root/'build/fclk-resume-20261101/sim-final-fixed-tail/golden-rtl'
assert (golden/'swd_core.v').is_file()
report=dict(hardwareAccess=False,targetClockRequestedHz=48000000,
            simulatedTargetPeriodNs=20.834,FCLK_Hz=133333333.33333333,
            returnDelayNs=16,rows=[])
for phase,dut in [('golden',golden),('candidate',root/'fpga/rtl')]:
    for label,delay,boundary in [('max66',0,0),('div33',1,1)]:
        work=out/phase
        args=[sys.executable,str(root/'tools/run_cm3_sim.py'),'--top','tb_cm3_swd',
              '--dut-rtl',str(dut),'--work-dir',str(work),
              'half_ns=3.75','target_clk_ns=20.833333','return_ns=16','phase=0',
              f'transfer_delay={delay}',f'rx_boundary={boundary}','benchmark']
        print('CM3',phase,label,flush=True)
        p=subprocess.run(args,cwd=root,capture_output=True)
        (out/(phase+'-'+label+'-runner.txt')).write_bytes(p.stdout+p.stderr)
        log=work/'xsim.log'
        if log.exists():shutil.copyfile(log,out/(phase+'-'+label+'-xsim.txt'))
        report['rows'].append(dict(phase=phase,label=label,halfPeriod=delay+1,
                                  sourceHashes={v.name:hashlib.sha256(v.read_bytes()).hexdigest() for v in dut.glob('*.v')},
                                  command=args,returnCode=p.returncode))
        (out/'result.json').write_text(json.dumps(report,indent=2)+'\n')
        print('CM3',phase,label,'exit',p.returncode,flush=True)
print('CM3 fixed clock runs completed; inspect first failures and matched conditions')
