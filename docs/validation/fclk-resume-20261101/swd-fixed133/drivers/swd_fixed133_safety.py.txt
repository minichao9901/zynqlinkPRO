from pathlib import Path
from datetime import datetime,timezone
import json,sys
root=Path(__file__).resolve().parents[2];sys.path.insert(0,str(root/'tools'))
from probe_flash import Session
from zynq_memory import PhysicalMemory
from test_probe_board import Memory
from dap_pipeline_bench import open_dap
import usb.util
out=root/'build/fclk-resume-20261101/swd-fixed133/safety';out.mkdir(exist_ok=False)
memory=PhysicalMemory(Session(out,Path('E:/Xilinx/SDK/2018.3'),'tcp:127.0.0.1:3121'))
addresses={'FCLK0_CTRL':0xf8000170,'FCLK1_CTRL':0xf8000180,'ARM_PLL':0xf8000100,
           'DDR_PLL':0xf8000104,'IO_PLL':0xf8000108,'SWD_STATUS':0x40010004,
           'PACKET_STATUS':0x42020004,'DDR_STATUS':0x40040008,'CDMA_STATUS':0x40020004}
values={n:memory.read_words(a)[0] for n,a in addresses.items()}
dev,tx,rx=open_dap()
try:dhcsr=Memory(tx,rx).word(0xe000edf0) # Read only; keep the current safe link profile.
finally:usb.util.dispose_resources(dev)
idle=not(values['SWD_STATUS']&1 or values['PACKET_STATUS']&15 or values['DDR_STATUS']&33) and bool(values['CDMA_STATUS']&2)
report=dict(recordedAtUtc=datetime.now(timezone.utc).isoformat(),purpose='Stop/handoff safety snapshot; no new test',
            registerWrites=False,targetWrites=False,registers={n:hex(v) for n,v in values.items()},
            DHCSR=hex(dhcsr),targetHalted=bool(dhcsr&(1<<17)),allObservedEnginesIdle=idle,
            note='The only remaining process expected is the pre-existing shared hw_server; no firmware/clock load performed here.')
(out/'result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
