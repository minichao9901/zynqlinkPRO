class TrackedMemory(Memory):
    def reset_counts(self):self.counts=dict(ok=0,returnedWait=0,returnedFault=0,returnedError=0,postedReadWords=0);self.last=[]
    def transfer(self,packet):
        reply=super().transfer(packet)
        if packet[0] in (5,6):
            status=reply[2] if packet[0]==5 and len(reply)>=3 else reply[3] if packet[0]==6 and len(reply)>=4 else 255
            key={1:'ok',2:'returnedWait',4:'returnedFault'}.get(status,'returnedError');self.counts[key]+=1
            if packet[0]==6 and packet[4]&3==3:self.counts['postedReadWords']+=packet[2]+256*packet[3]
            self.last=(self.last+[dict(requestHead=bytes(packet[:12]).hex(),replyHead=reply[:16].hex(),status=status)])[-8:]
        return reply

dev,tx,rx=open_dap();mem=TrackedMemory(tx,rx);mem.reset_counts();saved=None;initial=None
addr=0x20008000
try:
    setup(tx,rx,1000);initial=mem.word(0xe000edf0)
    report['target']=dict(CPUID=hex(mem.word(0xe000ed00)),deviceId=hex(mem.word(0xe0042000)),
                          DEMCR=hex(mem.word(0xe000edfc)),RCC=[mem.word(0x40021000),mem.word(0x40021004)])
    assert report['target']['CPUID']=='0x411fc231' and int(report['target']['deviceId'],16)&0xfff==0x414
    assert report['target']['RCC'][1]==5309450,'Target clock differs from the CDC48 Golden baseline'
    from elftools.elf.elffile import ELFFile
    report['targetFlashIdentity']=[]
    with (root/'prebuilt/golden/f103ze-cdc48.elf').open('rb') as f:
        for segment in ELFFile(f).iter_segments():
            if segment['p_type']=='PT_LOAD' and segment['p_filesz'] and 0x08000000<=segment['p_paddr']<0x08080000:
                data=segment.data();actual=mem.read(segment['p_paddr'],len(data));assert actual==data
                report['targetFlashIdentity'].append(dict(address=segment['p_paddr'],bytes=len(data),sha256=hashlib.sha256(actual).hexdigest()))
    if not initial&(1<<17):mem.write_word(0xe000edf0,0xa05f0003)
    saved=mem.read(addr,2048);(out/'sram-backup.bin').write_bytes(saved)
    report['backupSha256']=hashlib.sha256(saved).hexdigest();save()
    for rate in [1000000,16666667,22222222,33333333,66666666]:
        row=dict(requestedHz=rate,passed=False)
        mem.reset_counts()
        try:
            row['setupIdcode']=setup(tx,rx,rate//1000,speed_hz=rate).hex()
            diag=xfer(tx,rx,[0x80,255]);fclk,cfg=struct.unpack_from('<II',diag,4)
            row.update(FCLK_Hz=fclk,cfg=hex(cfg),**swd_clock_details(fclk,cfg))
            assert fclk==133333332
            for _ in range(128):
                r=mem.transfer([5,0,1,2]);assert r[:3]==bytes([5,1,1]) and int.from_bytes(r[3:7],'little')==0x1ba01477,r.hex()
            for _ in range(64):assert mem.word(0xe000ed00)==0x411fc231
            for loop in range(8):
                data=random.Random(0x133000+loop).randbytes(2048)
                mem.write(addr,data);assert mem.read(addr,2048)==data
            row.update(passed=True,dpIdReads=128,cpuidReads=64,patternRounds=8,bytesEachDirection=16384)
        except Exception as exc:
            row.update(error=repr(exc),lastTransactions=mem.last)
            try:row['queueAndCoreState']=xfer(tx,rx,[0x87]).hex()
            except Exception as err:row['diagnosticError']=repr(err)
        finally:

            row['internalWaitRetries']='Not observable in release RTL; returned WAIT is after retries'
            recover(dev,rx);setup(tx,rx,1000)

            assert row['restoreMatch']
            row['clockAfterRecovery']=live_clock_snapshot()
            assert row['clockAfterRecovery']['FCLK_Hz']==133333332 and row['clockAfterRecovery']['clockSource']==2
            clocks=physical.read_words(0xf8000100,33)

            print(json.dumps({k:row.get(k) for k in ['requestedHz','SWCLK_Hz','FCLK_Hz','passed','error']}),flush=True)
