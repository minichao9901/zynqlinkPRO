set root E:/zynqlinkPRO
set out $root/build/fclk-resume-20261101/swd-fixed133/sta
file mkdir $out
open_checkpoint $root/build/fclk-resume-20261101/release-133/system_wrapper_verified.dcp
# Tighten the passive pad-clock declaration to match the proposed /2 line
# rate. It does not clock protocol decisions. Keep the real 7 ns pad budget,
# IDDR half-cycle paths, input delays, CDC groups and all exceptions intact.
create_clock -name pad_swclk -period 15.000 -waveform {0 7.500} [get_ports swclk_0]
report_clocks -file $out/clocks.rpt
report_timing_summary -report_unconstrained -file $out/timing.rpt
report_timing -delay_type max -max_paths 30 -file $out/setup.rpt
report_timing -delay_type min -max_paths 30 -file $out/hold.rpt
report_timing -delay_type max -to [all_registers -async_pins] -max_paths 10 -file $out/recovery.rpt
report_timing -delay_type min -to [all_registers -async_pins] -max_paths 10 -file $out/removal.rpt
foreach sampler {u_swdio_iddr} {
  foreach phase {Q1 Q2} {
    set pin [get_pins -hier -filter "NAME =~ */$sampler/$phase"]
    report_timing -from $pin -delay_type max -max_paths 10 -file $out/setup_${phase}.rpt
    report_timing -from $pin -delay_type min -max_paths 10 -file $out/hold_${phase}.rpt
  }
}
check_timing -verbose -file $out/check_timing.rpt
report_clock_interaction -file $out/clock_interaction.rpt
report_exceptions -file $out/exceptions.rpt
set summary [report_timing_summary -return_string]
set met [string match {*All user specified timing constraints are met.*} $summary]
set fd [open $out/result.json w]
puts $fd "{\"analysisOnly\":true,\"FCLKPeriodNs\":7.5,\"proposedPadPeriodNs\":15,\"padMaxDelayNs\":7,\"allSpecifiedChecksPassed\":$met,\"newBitGenerated\":false,\"hardwareAccess\":false}"
close $fd
if {!$met} {error "Fixed-FCLK SWD supplement failed; no board test allowed"}
