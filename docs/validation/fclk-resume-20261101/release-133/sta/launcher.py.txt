from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,subprocess,sys
root=Path(__file__).resolve().parents[2];sys.path.insert(0,str(root))
from tools.run_cm3_sim import environment,VIVADO
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
original=json.loads((root/'build/fclk-resume-20261101/implementation-150-final-fixed-tail/job.json').read_text())
changed=[n for n,h in original['sources'].items() if n.startswith('fpga\\rtl\\') and sha(root/n)!=h]
if changed:raise RuntimeError('Routed RTL changed: '+str(changed))
dcp=root/'build/fclk-resume-20261101/final-routed-150.dcp'
assert sha(dcp)=='aa3897d8b421ec15431b7b56762376b313030bc3f2a05121bb70b49753112916'
out=root/'build/fclk-resume-20261101/release-133';out.mkdir(exist_ok=False)
env=environment();env['RDI_JAVAROOT']=(VIVADO/'tps/win64/jre9.0.4').as_posix()
options=dict(ZYNQLINK_RAW_TRACE='4',ZYNQLINK_ETM_DDR='1',ZYNQLINK_ETM_LANE_DELAY='1',ZYNQLINK_RTL_DEBUG='0',ZYNQLINK_IMPLEMENT_FCLK_HZ='133333333')
env.update(options);env['ZYNQLINK_WORK_DIR']=str(out/'runtime');(out/'runtime/tmp').mkdir(parents=True)
record=dict(source=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),
            routedRtlSource=original['source'],startedUtc=datetime.now(timezone.utc).isoformat(),
            inputDcp=str(dcp),inputDcpSha256=sha(dcp),options=options,
            freshBitGeneration=True,newPlacement=False,rtlSha256={p.name:sha(p) for p in (root/'fpga/rtl').glob('*.v')},
            verifierSha256=sha(root/'fpga/verify_routed.tcl'))
(out/'job.json').write_text(json.dumps(record,indent=2)+'\n')
with (out/'console.txt').open('w') as log:
    p=subprocess.run([sys.executable,str(root/'tools/run_vivado.py'),'-mode','batch','-source',str(root/'build/fclk-resume-20261101/release_133_verify.tcl'),'-log',str(out/'vivado.log'),'-journal',str(out/'vivado.jou')],env=env,cwd=root,stdout=log,stderr=subprocess.STDOUT)
record.update(returncode=p.returncode,finishedUtc=datetime.now(timezone.utc).isoformat())
for name in ['system_wrapper_verified.bit','verified_clock.json']:
    if (out/name).exists():record[name]=dict(sha256=sha(out/name),bytes=(out/name).stat().st_size)
(out/'job.json').write_text(json.dumps(record,indent=2)+'\n')
print('133MHz release verification/bitgen:',p.returncode)
raise SystemExit(p.returncode)
