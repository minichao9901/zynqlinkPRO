from pathlib import Path
import json,subprocess,sys
root=Path(__file__).resolve().parents[2]
out=root/'build/fclk-resume-20261101/board-133/protocols';out.mkdir(parents=True,exist_ok=True)
pair=root/'prebuilt/fclk133-release-20261101'
common=['--bit',str(pair/'probe.bit'),'--elf',str(pair/'application.elf')]


def run(label,args,allow_failure=False):
    print('Starting',label,flush=True)
    with (out/(label+'-console.txt')).open('w',encoding='utf8') as log:
        child=subprocess.run([sys.executable,*args],cwd=root,stdout=log,stderr=subprocess.STDOUT)
    print('Completed',label,'exit',child.returncode,flush=True)
    if child.returncode and not allow_failure:raise SystemExit(child.returncode)


def jobs(transport):
    run(transport+'-sram',['tools/benchmark_sram_matrix.py','--transport',transport,'--speeds','25000000,60000000','--seconds','0.3','--repeats','3','--output',str(out/(transport+'-sram.json')),*common])
    run(transport+'-peak-sram',['tools/h743_peak.py','--transport',transport,'--jtag-ir','4,5','--speed','60000','--address','0x20008000','--words','512','--depth','32','--loops','256','--repeats','3','--soak-seconds','10','--output',str(out/(transport+'-peak-sram.json')),*common])
    run(transport+'-rtt',['tools/benchmark_rtt_forward.py','--transport',transport,'--speeds','60000000','--chunks','1024','--seconds','2','--repeats','3','--output',str(out/(transport+'-rtt.json')),*common],allow_failure=True)
    if transport=='jtag':
        run('jtag-rtt25',['tools/benchmark_rtt_forward.py','--transport','jtag','--speeds','25000000','--chunks','1024','--seconds','2','--repeats','3','--output',str(out/'jtag-rtt25.json'),*common],allow_failure=True)
    run(transport+'-hss',['tools/benchmark_hss_matrix.py','--transports',transport,'--speeds','60000000','--seconds','0.5','--output',str(out/(transport+'-hss.json')),*common])


if '--jtag-child' in sys.argv:
    jobs('jtag')
else:
    jobs('swd')
    run('jtag-fixture',['tools/run_f103_jtag_fixture.py','--restore-swj','2','--output',str(out/'jtag-fixture.json'),'--',sys.executable,str(Path(__file__).resolve()),'--jtag-child'])
    print('All protocol jobs completed; inspect measurement acceptance fields',flush=True)
