from pathlib import Path
import subprocess,sys
root=Path(__file__).resolve().parents[2]
out=root/'build/fclk-resume-20261101/board-133/rtt-isolated/golden-load';out.mkdir(exist_ok=False)
pair=root/'prebuilt/golden'
jobs=[('load',['tools/run_xsct.py','tools/download_dap_fpga.tcl',str(pair/'application.elf'),str(pair/'probe.bit'),str(pair/'ps7_init.tcl')]),
      ('identity',['tools/verify_probe_ocm_code.py','--elf',str(pair/'application.elf'),'--out-dir',str(out/'identity')]),
      ('rtt',[str(root/'build/fclk-resume-20261101/rtt_etm_isolation.py'),'golden',str(pair)])]
for name,args in jobs:
    print('Starting Golden isolated',name,flush=True)
    with (out/(name+'.txt')).open('w') as log:p=subprocess.run([sys.executable,*args],cwd=root,stdout=log,stderr=subprocess.STDOUT)
    print('Finished Golden isolated',name,p.returncode,flush=True)
    if p.returncode:raise SystemExit(p.returncode)
