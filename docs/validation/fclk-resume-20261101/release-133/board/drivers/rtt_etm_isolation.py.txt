from pathlib import Path
import json,subprocess,sys,time
root=Path(__file__).resolve().parents[2];sys.path.insert(0,str(root/'tools'))
from dap_pipeline_bench import open_dap,setup
from test_probe_board import Memory
import usb.util
label=sys.argv[1];pair=root/sys.argv[2]
out=root/'build/fclk-resume-20261101/board-133/rtt-isolated'/label;out.mkdir(parents=True,exist_ok=False)
report=dict(passed=False,kind='controlled RTT with pre-existing ETM output disabled',
            targetClockWrites=False,FlashWrites=False,pair=str(pair))
def connect():
    d,o,i=open_dap();setup(o,i,50000);return d,Memory(o,i)
dev,mem=connect()
before=mem.word(0xe0041000);lock=mem.word(0xe0041fb4)
report.update(ETMCR_before=hex(before),ETMLSR_before=hex(lock),RCC_before=[mem.word(0x40021000),mem.word(0x40021004)])
try:
    if lock&3==3:mem.write_word(0xe0041fb0,0xc5acce55)
    mem.write_word(0xe0041000,before|0x400)
    for _ in range(100):
        if mem.word(0xe0041010)&2:break
        time.sleep(.001)
    else:raise RuntimeError('ETM failed to enter programming state')
    report['ETMCR_test']=hex(mem.word(0xe0041000))
    usb.util.dispose_resources(dev);dev=None
    args=[sys.executable,str(root/'tools/benchmark_rtt_forward.py'),'--transport','swd','--speeds','60000000',
          '--chunks','1024','--seconds','2','--repeats','3','--output',str(out/'rtt.json'),
          '--bit',str(pair/'probe.bit'),'--elf',str(pair/'application.elf')]
    report['command']=args
    with (out/'console.txt').open('w') as log:p=subprocess.run(args,cwd=root,stdout=log,stderr=subprocess.STDOUT)
    report['returnCode']=p.returncode
    report['passed']=p.returncode==0
finally:
    if dev is None:dev,mem=connect()
    if lock&3==3:mem.write_word(0xe0041fb0,0xc5acce55)
    mem.write_word(0xe0041000,before)
    report['ETMCR_after']=hex(mem.word(0xe0041000))
    if lock&3==3:mem.write_word(0xe0041fb0,0)
    report['ETM_state_restored']=report['ETMCR_after']==hex(before)
    report['RCC_after']=[mem.word(0x40021000),mem.word(0x40021004)]
    report['passed']=report['passed'] and report['ETM_state_restored'] and report['RCC_after']==report['RCC_before']
    usb.util.dispose_resources(dev)
    (out/'result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
if not report['passed']:raise SystemExit(1)
