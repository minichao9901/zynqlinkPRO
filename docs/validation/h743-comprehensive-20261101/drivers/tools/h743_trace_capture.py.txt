"""H743 ETMv4 continuous-window capture with the shipping DDR USB transport.

CoreSight sequencing follows the repository's web/app/target.js H7 path.
This runner preserves target PLLs and restores all managed configuration.
"""
import hashlib
import json
import struct
import time

from elftools.elf.elffile import ELFFile
from target_snapshot import snapshot

ETM, FUNNEL, ETF, TPIU = 0xe0041000, 0x5c013000, 0x5c014000, 0x5c015000


def capture(args, dev, tx, rx, mem, report):
    if (args.entry or args.reset_to_entry or args.usb_reconnect or args.ram_image
            or args.hclk_div != 1 or args.raw or args.ddr_physical or args.sync
            or args.transport == 'legacy'):
        raise ValueError('H743 native runner supports continuous DDR USB capture only; no F103-specific entry/reset/clock options')
    saved, locks = {}, {}
    oslocked = None
    initial = mem.word(0xe000edf0)
    clocks = snapshot(mem)
    report.update(target_profile='stm32h743', rcc_before=clocks, flash=[], transport='ddr-usb')

    def wait(address, mask, value):
        for _ in range(100):
            actual = mem.word(address)
            if actual & mask == value:
                return
            time.sleep(.005)
        raise RuntimeError(f'H743 handshake timeout at {address:#x}: {actual:#x}, expected {value:#x}/{mask:#x}')

    def put(address, value, verify=True):
        saved.setdefault(address, mem.word(address))
        mem.write_word(address, value)
        if verify and mem.word(address) != value:
            raise RuntimeError(f'H743 configuration readback mismatch: {address:#x}')

    def stop_fifo():
        if mem.word(ETF+0x20) & 1:
            put(TPIU+0x304, 3)
            put(ETF+0x304, mem.word(ETF+0x304) | 0x1000)
            put(ETF+0x304, mem.word(ETF+0x304) | 0x40, False)
            wait(ETF+0x304, 0x40, 0)
            wait(ETF+0xc, 4, 4)
        put(ETF+0x20, 0)
        wait(ETF+0xc, 4, 4)

    try:
        assert int(clocks['device'], 16) & 0xfff == 0x450
        with args.target_elf.open('rb') as stream:
            for segment in ELFFile(stream).iter_segments():
                address = segment['p_paddr']
                if segment['p_type'] == 'PT_LOAD' and segment['p_filesz'] and 0x08000000 <= address < 0x08200000:
                    data = mem.read(address, segment['p_filesz'])
                    assert data == segment.data(), f'H743 ELF/Flash mismatch at {address:#x}'
                    report['flash'].append(dict(address=address, bytes=len(data), sha256=hashlib.sha256(data).hexdigest()))
        assert report['flash'], 'No H743 Flash segments verified'
        mem.write_word(0xe000edf0, 0xa05f0003)
        wait(0xe000edf0, 1 << 17, 1 << 17)
        put(0x5c001004, mem.word(0x5c001004) | 0x700000)
        put(0xe000edfc, mem.word(0xe000edfc) | 0x1000000)
        put(0x580244e0, mem.word(0x580244e0) | 16)
        gpio = 0x58021000
        speed = args.gpio_speed if args.gpio_speed is not None else 2
        for address, mask, value in [(gpio, 0x3ff0, 0x2aa0), (gpio+4, 0x7c, 0),
                                     (gpio+8, 0x3ff0, speed*0x1550), (gpio+12, 0x3ff0, 0),
                                     (gpio+0x20, 0x0fffff00, 0)]:
            put(address, (mem.word(address) & ~mask) | value)
        for base in (ETM, FUNNEL, ETF, TPIU):
            locks[base] = mem.word(base+0xfb4)
            mem.write_word(base+0xfb0, 0xc5acce55)
            assert not mem.word(base+0xfb4) & 2
        oslocked = bool(mem.word(ETM+0x314) & 2)
        report['locks_before'] = {hex(base):hex(status) for base,status in locks.items()}
        report['oslock_before'] = oslocked
        mem.write_word(ETM+0x310, 0)
        wait(ETM+0x314, 2, 0)
        put(ETM+4, 0)
        wait(ETM+0xc, 1, 1)
        put(FUNNEL, 0x301)
        stop_fifo()
        put(ETF+0x28, 2); put(ETF+0x304, 3); put(ETF+0x20, 1)
        if not mem.word(TPIU+0x300) & 2:
            put(TPIU+0x304, mem.word(TPIU+0x304) | 0x1040, False)
            wait(TPIU+0x300, 2, 2)
        put(TPIU+0x204, 0); put(TPIU+4, 8); put(TPIU+0x400, 256, False); put(TPIU+0x304, 3)
        put(ETM+0x10, (mem.word(ETM+0x10) & ~0x1018) | 0x1010)  # Return stack and cycle count, no branch broadcast.
        put(ETM+0x38, 256)
        for offset, value in [(0x20, 0), (0x24, 0), (0x2c, 0), (0x40, args.trace_id),
                              (0x88, 0), (0x8c, 0), (0x80, 0x201)]:
            put(ETM+offset, value)
        put(ETM+4, 1)
        config = {'protocol': 'ETMv4'}
        for name, offset in dict(idr0=0x1e0,idr1=0x1e4,idr2=0x1e8,idr8=0x180,
                                 idr9=0x184,idr10=0x188,idr11=0x18c,idr12=0x190,
                                 idr13=0x194,config=0x10,traceid=0x40).items():
            config[name] = mem.word(ETM+offset)
        report.update(config=config, gpio_speed=speed, rcc_capture=snapshot(mem))
        (args.output/'target-before.json').write_text(json.dumps({hex(k):hex(v) for k,v in saved.items()},indent=2)+'\n')
        report['reference'] = struct.unpack('<4I',dev.ctrl_transfer(0xc0,0x61,0,0x4554,16))
        assert report['reference'][3] & 0x200, 'Probe lacks DDR USB capture'
        def resume():
            if args.ready_file:
                args.ready_file.write_text(json.dumps(dict(armed=True,unix=time.time(),monotonic=time.monotonic())))
            mem.write_word(0xe000edf0,0xa05f0001)
        from etm_ddr_usb_capture import capture as capture_ddr
        etm, ddr = capture_ddr(args.output,args.elf,dev,tx,rx,args.milliseconds/1000,
                               report['reference'][0],resume,args.trace_id,args.verify_ddr_physical)
        report['ddr'] = ddr
        if ddr.get('cleanupError') or not ddr.get('usbOwnerReleased'):
            report['usb_cleanup_error'] = ddr.get('cleanupError') or 'DDR USB owner was not released'
            raise RuntimeError(report['usb_cleanup_error'])
        report.update(ddr=ddr,quality=ddr['quality'],etm_bytes=len(etm),
                      etm_sha256=hashlib.sha256(etm).hexdigest(),capture_passed=True)
        (args.output/'etm.bin').write_bytes(etm)
    except BaseException as exc:
        report['capture_error'] = repr(exc)
        raise
    finally:
        try:
            if saved:
                mem.write_word(0xe000edf0,0xa05f0003)
                mem.write_word(0x5c001004,mem.word(0x5c001004)|0x700000)
                for base in locks:
                    mem.write_word(base+0xfb0,0xc5acce55)
                if ETM+4 in saved:
                    mem.write_word(ETM+0x310,0)
                    mem.write_word(ETM+4,0); wait(ETM+0xc,1,1)
                if ETF+0x20 in saved:
                    stop_fifo()
                last=[ETF+0x20,ETM+4,0x580244e0,0xe000edfc,0x5c001004]
                order=[a for a in reversed(saved) if a not in last]+[a for a in last[:2] if a in saved]
                for address in order:
                    mem.write_word(address,saved[address])
                report['restored'] = all(mem.word(a)==saved[a] for a in order)
                if oslocked is not None:
                    mem.write_word(ETM+0x310,int(oslocked))
                    report['oslock_restored'] = bool(mem.word(ETM+0x314)&2)==oslocked
                    assert report['oslock_restored'], 'H743 OS lock restoration mismatch'
                for base,status in locks.items():
                    if status&2:mem.write_word(base+0xfb0,0)
                report['locks_restored'] = all((mem.word(base+0xfb4)&2)==(status&2) for base,status in locks.items())
                assert report['locks_restored'], 'H743 CoreSight lock restoration mismatch'
                for address in last[2:]:
                    if address in saved:
                        mem.write_word(address,saved[address])
                        report['restored'] &= mem.word(address)==saved[address]
                report['rcc_after']=snapshot(mem)
                assert report['rcc_after']==clocks, 'H743 target clocks changed'
                assert report['restored'], 'H743 CoreSight restoration mismatch'
            if not initial&(1<<17):mem.write_word(0xe000edf0,0xa05f0001)
            report['run_halt_restored'] = bool(mem.word(0xe000edf0)&(1<<17))==bool(initial&(1<<17))
            assert report['run_halt_restored'], 'H743 run/halt state restoration mismatch'
            report['fpb_restored']=True  # This continuous path does not touch FPB.
        except BaseException as exc:
            report['target_cleanup_error'] = repr(exc)
            report['capture_passed'] = False
            raise
