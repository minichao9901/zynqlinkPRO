import sys,json
from pathlib import Path
from datetime import datetime,timezone
sys.path.insert(0,'tools')
import usb.util
from dap_pipeline_bench import open_dap,drain,xfer
from probe_stream import Probe
from benchmark_hpm5301 import setup,clock
from characterize_hpm6800_idle import set_idle
from riscv_target import RiscvTarget
from validate_fclk_performance import physical_snapshot

base=Path('build/hpm6800-idle-frontier-20261102')
report=dict(time_UTC=datetime.now(timezone.utc).isoformat(),before=physical_snapshot())
assert report['before']['allObservedEnginesIdle']
dev,out,inp=open_dap()
probe=Probe()
try:
    report['services']={hex(c):probe.action(c,2) for c in (0x31,0x32)}
    assert all(rc==0 and not info[0]&1 for rc,info in report['services'].values())
    drain(inp)
    setup(out,inp,10_000_000,0,connect=True)
    set_idle(out,inp,8)
    setup(out,inp,10_000_000,0)
    t=RiscvTarget(out,inp)
    t.active()
    report['dmstatus']=hex(t.dmi(0x11))
    assert int(report['dmstatus'],16)&(1<<10),'Target not running'
    setup(out,inp,25_000_000,0)
    report['clock']=clock(out,inp)
    report['idle_query']=xfer(out,inp,[0x8d,255]).hex()
    assert report['idle_query']=='8d000840'
finally:
    probe.close()
    usb.util.release_interface(dev,0)
    usb.util.dispose_resources(dev)
report['after']=physical_snapshot()
assert report['after']['allObservedEnginesIdle']
report['passed']=True
(base/'final-state.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
