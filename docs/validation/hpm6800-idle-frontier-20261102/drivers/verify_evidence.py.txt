from pathlib import Path
import hashlib,json,subprocess

root=Path.cwd()
dest=root/'docs/validation/hpm6800-idle-frontier-20261102'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def load(p):return json.loads(p.read_text(encoding='utf-8'))
def raw(name):return load(dest/'raw'/name/'result.json')
for name in ('hpm6800-idle64-120-control-20261102','hpm6800-idle64-133-experimental-20261102'):
    base=root/'prebuilt'/name
    m=load(base/'manifest.json')
    for rel,entry in m['files'].items():
        p=base/rel
        assert sha(p)==entry['sha256'] and p.stat().st_size==entry['bytes'],p
    for name,h in m['rtl_build_input_sha256'].items():
        assert sha(root/'fpga/rtl'/name)==h,name
    for name,h in m['ps_build_input_sha256'].items():
        candidates=[p for p in (root/'firmware/xusbps_daplink_dma').rglob(name) if p.is_file()]
        assert any(sha(p)==h for p in candidates),(name,h)
for name,h in load(dest/'build-inputs/sha256.json').items():
    assert sha(root/name)==h and sha(dest/'build-inputs'/(name+'.txt'))==h,name
for p in (dest/'raw').glob('*/result.json'):
    d=load(p)
    assert not d.get('restore_error'),(p,d.get('restore_error'))
    if 'restore' in d:assert all(d['restore'].values()),p
    if 'restore_match' in d:assert d['restore_match'],p
    for r in d.get('rows',[]):
        if 'restore_match' in r:assert r['restore_match'],p
assert all(r['passed'] for r in raw('final133-sram-stress')['rows'])
assert all(r['passed'] and r['sequence_exact'] and r['all_published_received'] for r in raw('final133-rtt')['rows'])
for name in ('final133-hss-margin','final133-hss-multi'):
    assert all(r['skipped']==0 and r['errors']==0 and r['tail_count_delta']==0 and r['all_data_frames_verified'] for r in raw(name)['rows']),name
assert [r['skipped'] for r in raw('final133-hss-single')['rows']]==[0,1268,0]
assert raw('final133-legacy25')['rows'][0]['passed']
assert raw('identity-final-133')['passed']
assert load(dest/'raw/final-state.json')['passed']
summary=dict(scope='single-board HPM6800 JTAG over-spec exploration; final133 timing; not full product acceptance',
    sram=raw('final133-sram-stress')['rows'],rtt=raw('final133-rtt')['rows'],
    hss={name:raw(name)['rows'] for name in ('final133-hss-single','final133-hss-margin','final133-hss-multi','final133-hss-overload-powered')},
    final_state=load(dest/'raw/final-state.json'),
    matrix_observations=len(load(dest/'matrix-summary.json')['rows']))
(dest/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
# Evidence index excludes itself and source pin to avoid self-referential hashes.
index=dict(files={str(p.relative_to(dest)).replace('\\','/'):dict(bytes=p.stat().st_size,sha256=sha(p))
    for p in sorted(dest.rglob('*')) if p.is_file() and p.name not in ('evidence.json','source-revision.json')})
(dest/'evidence.json').write_text(json.dumps(index,indent=2)+'\n')
old=root/'docs/validation/hpm6800-acceptance-20261102'
for rel,e in load(old/'evidence.json')['files'].items():
    assert sha(old/rel)==e['sha256'],('R19',rel)
print('All pair payloads, build inputs, restores, final measurements and R19 evidence hashes verified.')
