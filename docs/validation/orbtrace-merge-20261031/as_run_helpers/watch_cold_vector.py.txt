from pathlib import Path
import sys
sys.path.insert(0,'tools')
from probe_flash import Session
p=Path('docs/validation/orbtrace-qspi-20261031/cold_power_cycle/vector_watch');p.mkdir(exist_ok=False)
s=Session(p,Path('E:/Xilinx/SDK/2018.3'),'tcp:127.0.0.1:3121')
print(s.run('''rst -system
set bp [bpadd -addr 0 -type hw -mode 2]
puts "WATCH_ID $bp"
for {set i 0} {$i < 4} {incr i} {
 set rc [catch {con -block -timeout 5} why]
 puts "WATCH_ITER $i RC $rc WHY $why"
 puts [rrd pc]
 puts [rrd r0]
 puts [rrd r1]
 puts [rrd r2]
 puts [rrd r3]
 puts [mrd -force 0 4]
 if {$rc} {break}
}
bpremove $bp
puts WATCH_REMOVED
catch {con}
puts [targets]
'''))
