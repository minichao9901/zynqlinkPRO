from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,statistics,subprocess,difflib

root=Path(__file__).resolve().parents[2]
base=Path(__file__).resolve().parent
other=Path('E:/superlinkPRO')
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,d):p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n',encoding='utf8')
data={}
for name in ['legacy','current']:
    report=json.loads((base/(name+'-bench/results.json')).read_text(encoding='utf8'))
    assert len(report['rows'])==3 and report['inputUnchanged']
    rows=report['rows'];summary=rows[0]['summary']
    phases=rows[0]['performance']['decodePhasesSeconds']
    store=rows[0]['performance']['storePhasesSeconds']
    data[name]=dict(inputSha256=report['inputSha256'],sourceRecord=report['sourceRecord'],
        runs=3,inputUnchanged=True,summary=summary,
        completeProcessingSeconds=[r['completeProcessingWallSeconds'] for r in rows],
        decodeSeconds=[r['performance']['decodeSeconds'] for r in rows],
        completeProcessingMedian=statistics.median(r['completeProcessingWallSeconds'] for r in rows),
        decodeMedian=statistics.median(r['performance']['decodeSeconds'] for r in rows),
        phaseMedians={k:statistics.median(r['performance']['decodePhasesSeconds'][k] for r in rows) for k in phases},
        storePhaseMedians={k:statistics.median(r['performance']['storePhasesSeconds'][k] for r in rows) for k in store},
        ETMBytes=rows[0]['files']['capture.bin'],nativeBinBytes=rows[0]['files']['native.bin'],
        allSummaryAndSharedQualityMatched=all(r['summaryMatchesLegacy'] and r['sharedQualityMatchesLegacy'] for r in rows),
        allNoLargeJsonl=all(not r['largeJsonIntermediates'] for r in rows))
    folder=base/(name+'-bench/records')/rows[0]['record']
    manifest=json.loads((folder/'store/manifest.json').read_text(encoding='utf8'))
    data[name]['engine']=manifest['engine']['name']
    data[name]['fallback']=manifest['engine']['fallback']
fingerprints=[]
for p in sorted((root/'etm-service/build-native').glob('*.exe')):
    q=other/'etm/build-native'/p.name
    if q.exists():fingerprints.append(dict(name=p.name,currentSha256=digest(p),superlinkSha256=digest(q),equal=digest(p)==digest(q)))
sources=[]
for name in ['native_pipeline.py','native_storebuild.py','native_facts_index.py','trace_store_index.py',
             'trace_store_write.py','decode_stream.py','decode_trace.py']:
    p=root/'etm-service/tools'/name;q=other/'etm/tools'/name
    sources.append(dict(name=name,currentSha256=digest(p),superlinkSha256=digest(q),equal=p.read_bytes()==q.read_bytes()))
diff=''.join(difflib.unified_diff((other/'etm/tools/native_storebuild.py').read_text(encoding='utf8').splitlines(True),
    (root/'etm-service/tools/native_storebuild.py').read_text(encoding='utf8').splitlines(True),
    fromfile='superlinkPRO/native_storebuild.py',tofile='zynqlinkPRO/native_storebuild.py'))
(base/'native-storebuild-difference.txt').write_text(diff,encoding='utf8')
ratio=dict(ETMBytes=data['current']['ETMBytes']/data['legacy']['ETMBytes'],
    instructions=data['current']['summary']['instructions']/data['legacy']['summary']['instructions'],
    nativeBinBytes=data['current']['nativeBinBytes']/data['legacy']['nativeBinBytes'],
    completeProcessing=data['current']['completeProcessingMedian']/data['legacy']['completeProcessingMedian'])
historical=json.loads((root/'docs/validation/etm-e2e-clock-20261101/comparison.json').read_text(encoding='utf8'))
result=dict(recordedAtUtc=datetime.now(timezone.utc).isoformat(),hardwareAccess=False,
    sourceHead=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),
    superlinkHead=subprocess.check_output(['git','rev-parse','HEAD'],cwd=other,text=True).strip(),
    scope='Fresh offline full API processing through native decode/store/index/browse preparation; no acquisition or browser rendering',
    legacyClock=dict(RCC_CFGR=5308570,HCLK_Hz=12000000,TRACECLK_Hz=6000000),
    currentClock=dict(RCC_CFGR=5309450,HCLK_Hz=48000000,TRACECLK_Hz=24000000),
    benchmarks=data,scaleRatios=ratio,binaryComparison=fingerprints,sourceComparison=sources,
    prior12RunDecodePhases={k:statistics.median(r['nativePhases'][k] for r in historical['rows']) for k in historical['rows'][0]['nativePhases']},
    limitations=['OS file cache not cleared; sequential services, not simultaneous performance runs',
        'Same-byte old input retest does not prove all captures/platforms meet <=10s',
        'Superlink H743 1s release report measured 14.109..14.350s processing, target <=10s was not passed',
        'No source optimization, board, clock, PLL, Flash or FPGA/PS artifact change'])
assert len(fingerprints)==5 and all(r['equal'] for r in fingerprints)
save(base/'audit.json',result)
print('old API/decode medians',data['legacy']['completeProcessingMedian'],data['legacy']['decodeMedian'])
print('new API/decode medians',data['current']['completeProcessingMedian'],data['current']['decodeMedian'])
print('scale ratios',ratio)
