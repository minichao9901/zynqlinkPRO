import {readFile,writeFile,rename} from 'node:fs/promises';
import {pathToFileURL} from 'node:url';
import path from 'node:path';
const meta=JSON.parse(await readFile('C:/Users/Administrator/.codex/validation/current-h743-live-run.json','utf8'));
const apiBase=`http://127.0.0.1:${meta.webPort}/etm/api/`,usbBase=`http://127.0.0.1:${meta.usbPort}/`;
const sleep=ms=>new Promise(resolve=>setTimeout(resolve,ms));
async function usb(op,more={}){
  const response=await fetch(usbBase,{method:'POST',headers:{Authorization:'Bearer '+meta.usbToken},body:JSON.stringify({op,...more})});
  const data=await response.json();if(data.error)throw Error(data.error);return data.value;
}
async function api(route,body){
  const response=await fetch(apiBase+route,body===undefined?{}:{method:'POST',...(body instanceof FormData?{body}:{headers:{'Content-Type':'application/json'},body:JSON.stringify(body)})});
  const data=await response.json();if(!response.ok)throw Error(JSON.stringify(data));return data;
}
async function gate(send,stage){
  const file=path.join(meta.data,'load-gate.json'),temporary=file+'.new';
  await writeFile(temporary,JSON.stringify({send,stage,time:Date.now()/1000}));
  for(let attempt=0;;attempt++){try{await rename(temporary,file);break;}catch(error){if(attempt>=49||!['EPERM','EBUSY','EACCES'].includes(error.code))throw error;await sleep(5);}}
}
const {WebUsbDapProbe}=await import(pathToFileURL(path.join(meta.worktree,'web/app/rtt/dap-webusb.js')));
const {EtmTarget}=await import(pathToFileURL(path.join(meta.worktree,'etm/web/app/target.js')));
const probe=new WebUsbDapProbe(),target=new EtmTarget(message=>console.log('[target]',message),{host:{}});
probe.device={
  transferOut:async(_ep,data)=>({bytesWritten:await usb('write',{bytes:[...new Uint8Array(data.buffer??data,data.byteOffset??0,data.byteLength)]})}),
  transferIn:async()=>{const bytes=Uint8Array.from(await usb('read'));return{data:new DataView(bytes.buffer)};},
};
Object.assign(probe,{pkt:512,maxWords:120,epOut:2,epIn:1,_framing:'short',_ready:true,skipClearHalt:true,onLog:message=>console.log('[probe]',message)});
target.probe=probe;target.hseHz=25e6;target.profile='stm32h743';
const label=process.env.RUN_LABEL||'initial-200ms';
const durations=(process.env.DURATIONS||'.2').split(',').map(Number);
const repeats=Number(process.env.REPEATS||1),results=[];
let success=false;
const watchdog=setTimeout(()=>{console.error('hardware harness watchdog');process.exit(9);},1800000);
try{
  await gate(false,'configure');await sleep(300);
  await probe._targetInit({clock:1000000});const identified=await target.inspect();
  if(identified.profile!=='stm32h743')throw Error('Unexpected target '+identified.profile);
  console.log('Identified',identified.profile,JSON.stringify(identified.clocks));
  const form=new FormData();form.append('elf',new Blob([await readFile(meta.elf)]),'fw.elf');
  const elf=await api('elf',form);await target.verifyElf(elf);
  const sources={};
  for(const filename of elf.sourceFiles||[]){
    const normalized=path.resolve(filename),relative=path.relative(meta.worktree,normalized);
    if(relative.startsWith('..')||path.isAbsolute(relative))continue;
    try{sources[filename]=await readFile(normalized,'utf8');}catch{}
  }
  for(const seconds of durations){
    for(let round=0;round<repeats;round++){
      const stage=`${label}-${Math.round(seconds*1000)}ms-${round+1}`;
      await gate(false,stage+'-preparing');await sleep(300);
      const snapshot=await target.configure({traceid:2,branchBroadcast:false,returnStack:true,cycleCount:true,pinSpeed:Number(process.env.PIN_SPEED||3)});
      console.log('Configured',stage,JSON.stringify(snapshot.config));
      const continuous=process.env.CONTINUOUS==='1';
      if(continuous){
        const before=JSON.parse(await readFile(path.join(meta.data,'serial-state.json'),'utf8'));
        if(Date.now()/1000-before.time>1)throw Error('Serial load heartbeat stale before capture');
        await target.run();await gate(true,stage+'-warm');await sleep(350);
        const warm=JSON.parse(await readFile(path.join(meta.data,'serial-state.json'),'utf8'));
        if(Date.now()/1000-warm.time>1||warm.received-before.received<5)throw Error('Serial echo load is not responding; capture refused');
      }
      const started=Date.now()/1000;
      const created=await api('capture',{
        rate:500000000,seconds,threshold:1.65,sync:!continuous,decode:true,clock:0,data:[1,2,3,4],
        fraction:Number(process.env.FRACTION||.25),glitchNs:Number(process.env.GLITCH_NS||4),rawFormat:'kvdat',config:snapshot.config,elfId:elf.elfId,
        targetVerified:true,targetSnapshot:snapshot,captureScope:'continuous-banner-echo-acceptance',
        libraryOrigin:'validation',sources,
      });
      const row={stage,seconds,round:round+1,record:created.record,job:created.job,started,states:[],snapshot};
      let resumed=continuous,last='',finished=false,loadPaused=false;
      while(Date.now()/1000-started<600){
        const job=await api('jobs/'+created.job);
        if(job.state==='armed'&&!row.armedObserved)row.armedObserved=Date.now()/1000;
        if(continuous&&row.armedObserved&&!loadPaused&&Date.now()/1000-row.armedObserved>=seconds+.25){await gate(false,stage+'-sample-window-finished');loadPaused=true;row.loadPaused=Date.now()/1000;}
        if(job.state!==last){row.states.push({state:job.state,time:Date.now()/1000});console.log(stage,job.state);last=job.state;}
        if(job.state==='armed'&&!resumed){
          row.resumeRequested=Date.now()/1000;await target.run();row.resumeConfirmed=Date.now()/1000;
          await gate(true,stage);resumed=true;
        }
        if(resumed&&!loadPaused&&['saving','decoding','done','error','cancelled'].includes(job.state)){
          await gate(false,stage+'-processing');loadPaused=true;row.loadPaused=Date.now()/1000;
        }
        if(['done','error','cancelled'].includes(job.state)){
          row.finalJob=job;row.finished=Date.now()/1000;finished=true;break;
        }
        await sleep(job.state==='queued'||job.state==='starting'?30:job.state==='armed'?20:100);
      }
      if(!finished)throw Error('Capture job timed out');
      if(!resumed)throw Error('Acquisition completed before target was resumed');
      results.push(row);await writeFile(path.join(meta.evidence,label+'-results.json'),JSON.stringify(results,null,2));
      if(row.finalJob.state!=='done')throw Error('Capture failed: '+row.finalJob.error);
      row.overview=await api('browse/'+created.record+'/overview');row.session=await api('records/'+created.record);
      row.serialState=JSON.parse(await readFile(path.join(meta.data,'serial-state.json'),'utf8'));
      if(Date.now()/1000-row.serialState.time>1)throw Error('Serial load heartbeat lost; do not accept this capture');
      const q=row.overview.quality||{};
      console.log('Quality',JSON.stringify(q));
      let onlyTerminalPacket=false;
      if(q.packetIssues){
        const first=await api('browse/'+created.record+'/issues?limit=1');
        const tail=await api('browse/'+created.record+'/issues?offset='+Math.max(0,first.total-30)+'&limit=30');
        const packets=(tail.rows||[]).filter(x=>x.category==='packet');
        onlyTerminalPacket=packets.length===q.packetIssues&&packets.every(x=>x.reason==='包问题：I_INCOMPLETE_EOT'&&x.offset+x.bytes===q.inputBytes);
        row.terminalBoundaryPackets=packets;
      }
      row.qualityAccepted=(!q.packetIssues||onlyTerminalPacket)&&!q.overflowPackets&&!q.unprocessedBytes&&!q.unaccountedBytes&&!q.overlapBytes&&!q.decoderLastError;
      await writeFile(path.join(meta.evidence,label+'-results.json'),JSON.stringify(results,null,2));
      if(!row.qualityAccepted)throw Error('Trace quality gate failed; keep raw capture and inspect before extending window');
      console.log('PASS',stage,'instructions',row.overview.summary.instructions,'pipeline',row.overview.performance?.captureToDecodedSeconds);
    }
  }
  success=true;
}finally{
  clearTimeout(watchdog);await gate(false,'idle').catch(()=>{});
  if(success){await target.halt();await target.restore();await target.run();}
  else{try{await target.halt();}catch{}console.log('Failure: target left halted for diagnosis; no automatic resume');}
  target.probe=null;
}
