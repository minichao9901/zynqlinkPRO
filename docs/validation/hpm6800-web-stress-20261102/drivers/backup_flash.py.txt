from pathlib import Path
import subprocess,json,time,hashlib
b=Path(__file__).resolve().parent
dest=b/'target-backup';dest.mkdir(exist_ok=False)
args=['E:/sdk_env_v1.11.0/tools/openocd/openocd.exe',
 '-s','E:/sdk_env_v1.11.0/tools/openocd/tcl','-s','E:/sdk_env_v1.11.0/hpm_sdk/boards/openocd','-f',str(b/'hpm6800.cfg'),
 '-c','init','-c','halt','-c','riscv set_mem_access progbuf',
 '-c','dump_image {'+(dest/'flash-original.bin').as_posix()+'} 0x80000000 65536',
 '-c','dump_image {'+(dest/'flash-original-repeat.bin').as_posix()+'} 0x80000000 65536',
 '-c','reg pc','-c','resume','-c','shutdown']
started=time.perf_counter()
r=subprocess.run(args,capture_output=True,text=True,encoding='utf-8',errors='replace',timeout=180)
(b/'backup-flash.log').write_text(r.stdout+r.stderr,encoding='utf-8')
report=dict(argv=args,returncode=r.returncode,seconds=time.perf_counter()-started,
 target_flash_writes=False,method='halted CPU progbuf reads; no SBA XIP reads; no reset/init-clock/DDR hooks')
if not r.returncode:
    data=(dest/'flash-original.bin').read_bytes()
    report.update(bytes=len(data),sha256=hashlib.sha256(data).hexdigest(),repeat_matches=data==(dest/'flash-original-repeat.bin').read_bytes())
(b/'backup-flash.json').write_text(json.dumps(report,indent=2)+'\n')
assert r.returncode==0,(r.returncode,r.stdout[-1000:],r.stderr[-1000:])
assert report['bytes']==65536 and report['repeat_matches'],report
print(report['sha256'])
