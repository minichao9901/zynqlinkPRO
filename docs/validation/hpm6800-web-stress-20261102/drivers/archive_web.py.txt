"""Archive selected public fixtures/evidence; never runtime tokens or original Flash."""
from pathlib import Path
import shutil,json,hashlib
ROOT=Path.cwd();BASE=ROOT/'build/hpm6800-web-stress-20261102'
DEST=ROOT/'docs/validation/hpm6800-web-stress-20261102'
PAIR=ROOT/'prebuilt/hpm6800-web-133-20261102'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def load(p):return json.loads(p.read_text(encoding='utf-8'))
def save(p,v):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
secrets=[p.read_text().strip().encode() for p in BASE.rglob('cli-token')]
def copy(p,q):
    if any(secret in p.read_bytes() for secret in secrets):raise RuntimeError('Token in selected evidence: '+str(p))
    q.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(p,q)
PAIR.mkdir(exist_ok=True);DEST.mkdir(exist_ok=True)
for p in (BASE/'pair-web1').iterdir():
    if p.is_file():copy(p,PAIR/p.name)
for d in (PAIR,DEST):(d/'.gitattributes').write_text('* -text -whitespace\n')
sources=['probe_stream.c','probe_stream.h','probe_target.c','xusbps_intr_example.c','DAP/Source/DAP.c','DAP/Include/ps_fclk.h']
build=load(PAIR/'ps-build.json')
for name,expected in build['source_sha256'].items():
    p=ROOT/'firmware/xusbps_daplink_dma'/name
    assert sha(p)==expected,('PS source changed since actual build',p)
for name in sources:copy(ROOT/'firmware/xusbps_daplink_dma'/name,DEST/'build-inputs'/name)
save(DEST/'build-inputs/manifest.json',dict(base_revision='b5bb054a7d6b8f007cfe96f0302f12f03357eb4e',
    files={name:sha(DEST/'build-inputs'/name) for name in sources},
    note='Exact modified PS inputs, before Git text normalization; overlay on base. Full build-input hashes in prebuilt ps-build.json.'))
excluded={'runtime','runtime-web1','runtime-v2','runtime-pressure','frozen-data','ps-web1','pair-web1','target-backup','__pycache__'}
for d in BASE.iterdir():
    if not d.is_dir() or d.name in excluded:continue
    for p in d.rglob('*'):
        if not p.is_file() or '__pycache__' in p.parts:continue
        if p.name in ('readback-private.bin','flash-original.bin','flash-original-repeat.bin'):continue
        if p.suffix not in ('.json','.txt','.log','.jsp','.csv','.bin','.tcl'):continue
        copy(p,DEST/'raw'/p.relative_to(BASE))
for p in BASE.iterdir():
    if not p.is_file():continue
    if p.name.startswith('runtime') or 'private' in p.name:continue
    if p.suffix in ('.json','.log'):copy(p,DEST/'logs'/p.name)
    elif p.suffix in ('.py','.cfg'):copy(p,DEST/'drivers'/(p.name+'.txt'))
for name in ('validate_hpm6800_web.py','stress_hpm6800_web.py'):
    copy(ROOT/'tools'/name,DEST/'drivers'/(name+'.txt'))
copy(BASE/'frozen-flash-proof-full.png',DEST/'frozen-flash-proof.png')
copy(ROOT/'build/bridge-package/latest-verification.json',DEST/'package-offline-verification.json')
copy(ROOT/'dist/zynq-etm-bridge/package-manifest.json',DEST/'package-manifest.json')
copy(ROOT/'dist/zynq-etm-bridge/package-check.json',DEST/'package-check.json')
manifest=dict(status='HPM6800 native speed and Web functional evidence; remaining ZL-005 missing DEF and ZL-094 CDC gaps, not a universal release',
    formal_product_release=False,source_revision='pending final source commit',source_variant_base='b5bb054',
    source_overlay='../../docs/validation/hpm6800-web-stress-20261102/build-inputs/manifest.json',
    bit_reused_unchanged_from='prebuilt/hpm6800-idle64-133-experimental-20261102/probe.bit',
    build=build['options'],timing=load(PAIR/'verified_clock.json'),
    evidence='../../docs/validation/hpm6800-web-stress-20261102/README.md',
    files={p.relative_to(PAIR).as_posix():dict(bytes=p.stat().st_size,sha256=sha(p))
        for p in PAIR.rglob('*') if p.is_file() and p.name not in ('manifest.json','README.md')})
save(PAIR/'manifest.json',manifest)
(PAIR/'README.md').write_text('''# HPM6800EVK Web + 66.667MHz实验配对

PL按133.333333MHz/7.5ns验证，复用R20的bit；PS增加RISC-V最高档、启动诊断及旧HID架构/状态兼容。不是全平台Golden，不固化探针QSPI。

实际Web烧录/RTT/调试及随机切换使用此配对；同配对重载后原生SRAM约3.44/3.40MB/s、RTT3.19MB/s、HSS296.296kS/s对照通过。Web默认烧录实际10MHz。J-Scope跨功能缺首DEF（ZL-005）仍复现，另有随机CDC字节缺口ZL-094，不能称全通过。

见[完整报告](../../docs/validation/hpm6800-web-stress-20261102/README.md)、[manifest](manifest.json)。PS输入为b5bb054加报告build-inputs精确覆盖，完整输入SHA在ps-build.json；提交后source_revision注明承载修正的源码提交。
''',encoding='utf-8')
save(DEST/'evidence.json',dict(files={p.relative_to(DEST).as_posix():dict(bytes=p.stat().st_size,sha256=sha(p))
    for p in DEST.rglob('*') if p.is_file() and p.name!='evidence.json'}))
print('Archived',len(list(DEST.rglob('*'))),'entries; original/readback Flash and runtime tokens excluded.')
