"""Same probe, ELF, erase/program and byte-exact XIP readback at 10MHz."""
from pathlib import Path
import sys,json,time,re,subprocess,hashlib
from elftools.elf.elffile import ELFFile
sys.path.insert(0,str(Path.cwd()/'tools'))
from probe_build_info import live_clock_snapshot
b=Path(__file__).resolve().parent
elf=Path('firmware/target/hpm6800_web_validation/scope/fw.elf').resolve()
dest=b/'openocd-scope-actual10';dest.mkdir(exist_ok=False)
with elf.open('rb') as f:
    segments=sorted((int(s['p_paddr']),s.data()) for s in ELFFile(f).iter_segments() if s['p_type']=='PT_LOAD' and s['p_filesz'] and 0x80000000<=s['p_paddr']<0x81000000)
merged=[]
for addr,data in segments:
    if merged and merged[-1][0]+len(merged[-1][1])==addr:merged[-1][1]+=data
    else:merged.append([addr,data])
report=dict(elf=str(elf),elf_sha256=hashlib.sha256(elf.read_bytes()).hexdigest(),load_bytes=sum(len(d) for a,d in merged),
    clock_khz=10000,verify='CPU progbuf XIP dumps, exactly the same merged ELF regions as Web; every byte compared on host',rows=[])
def save():(dest/'result.json').write_text(json.dumps(report,indent=2)+'\n')
for repeat in range(4):
    folder=dest/('warmup' if repeat==0 else f'round-{repeat}');folder.mkdir()
    commands=['set bench0 [ms]','init','halt','echo [format {BENCH_PREP_MS %d} [expr {[ms]-$bench0}]]','set phase [ms]',
        'flash write_image erase {'+elf.as_posix()+'}',
        'echo [format {BENCH_PROGRAM_MS %d} [expr {[ms]-$phase}]]','riscv set_mem_access progbuf','set phase [ms]']
    for i,(addr,data) in enumerate(merged):commands.append('dump_image {'+(folder/f'verify-{i}.bin').as_posix()+f'}} {addr:#x} {len(data)}')
    commands+=['echo [format {BENCH_VERIFY_MS %d} [expr {[ms]-$phase}]]','set phase [ms]','reset run',
        'echo [format {BENCH_RESET_MS %d} [expr {[ms]-$phase}]]','echo [format {BENCH_TOTAL_MS %d} [expr {[ms]-$bench0}]]','shutdown']
    args=['E:/sdk_env_v1.11.0/tools/openocd/openocd.exe','-s','E:/sdk_env_v1.11.0/tools/openocd/tcl',
        '-s','E:/sdk_env_v1.11.0/hpm_sdk/boards/openocd','-f',str(b/'hpm6800.cfg')]
    for c in commands:args+=['-c',c]
    row=dict(repeat=repeat,warmup=repeat==0,argv=args,passed=False)
    report['rows'].append(row);save();started=time.perf_counter();lines=[]
    process=subprocess.Popen(args,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,encoding='utf-8',errors='replace',creationflags=subprocess.CREATE_NO_WINDOW)
    for line in process.stdout:
        lines.append(line)
        if repeat==0 and 'BENCH_PREP_MS ' in line:row['live_clock']=live_clock_snapshot()
    process.wait(timeout=180);row['seconds']=time.perf_counter()-started;row['returncode']=process.returncode
    log=''.join(lines);(folder/'openocd.log').write_text(log,encoding='utf-8')
    row['timings_ms']={k.lower():int(v) for k,v in re.findall(r'BENCH_(PREP|PROGRAM|VERIFY|RESET|TOTAL)_MS (\d+)',log)}
    row['regions']=[dict(addr=hex(a),bytes=len(d),sha256=hashlib.sha256((folder/f'verify-{i}.bin').read_bytes()).hexdigest(),
        byte_exact=(folder/f'verify-{i}.bin').read_bytes()==d) for i,(a,d) in enumerate(merged)] if process.returncode==0 else []
    row['passed']=process.returncode==0 and len(row['regions'])==len(merged) and all(x['byte_exact'] for x in row['regions'])
    save();print(repeat,row['seconds'],row['timings_ms'],row['passed'],flush=True)
    assert row['passed'],row
    if repeat==0:assert abs(row['live_clock']['actual_link_Hz']-10000000)<10,row['live_clock']
report['passed']=True;save()
